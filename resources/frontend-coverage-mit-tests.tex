% Qualitative Bewertung
\subsection*{Qualitative Test-Analyse}

\textbf{Gut getestet:}
\begin{itemize}[noitemsep]
	\item \textbf{Auth \& Guards:} Umfassende Tests für sicherheitskritische Funktionen (Token-Refresh, 401-Handling, Reauth-Loop-Breaker)
	\item \textbf{Shared/Validation:} Server-Validation-Interceptor mit Edge-Cases (PascalCase-Mapping, Toast-Skipping)
	\item \textbf{Search-Strategie:} Gute Test-Dichte für Filter-Rules und Grid-Konfiguration
\end{itemize}

\textbf{Verbesserungspotenzial:}
\begin{itemize}[noitemsep]
	\item \textbf{Experiment-Wizard:} Nur 21 Tests für fast 6.000 Zeilen Code -- kritische Beam-Calculations und Panel-Logik unzureichend abgedeckt
	\item \textbf{Services:} Domain-Services (ExperimentService, ParameterService) haben Lücken bei API-Error-Handling
	\item \textbf{Dashboard:} Einstiegspunkt mit niedriger Coverage -- Grid-Interaktionen fehlen
\end{itemize}

\textbf{E2E-Test-Lücken:}
\begin{itemize}[noitemsep]
	\item Nur 6 Playwright-Tests für gesamte User-Journeys
	\item Empfehlung: Kritische Flows wie \textit{Experiment-Import}, \textit{Beam-Calculation-Workflow}, \textit{Parameter-Copy} sollten End-to-End getestet werden
\end{itemize}
